cmake_minimum_required(VERSION 3.0)
project(DMAVIDEO LANGUAGES CXX)


#стандарт
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -D_M_IX86")


#библиотеки DMAGLIDE
file(GLOB DMAVIDEO_SOURCE "*.cpp")
file(GLOB DMAVIDEO_HEADERS "*.h")

#lib
add_library(DMAVIDEO STATIC ${DMAVIDEO_SOURCE} ${DMAVIDEO_HEADERS})
#dll
add_library(DMAVIDEO_SHARED SHARED ${DMAVIDEO_SOURCE} ${DMAVIDEO_HEADERS})
set_target_properties(DMAVIDEO_SHARED PROPERTIES OUTPUT_NAME "DMAVIDEO")
